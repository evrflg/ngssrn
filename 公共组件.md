## 下拉刷新组件

```js
import { PullToRefresh } from "@/components/common/PullToRefresh";

const [items, setItems] = useState(Array.from({ length: 20 }).map((_, i) => `项目 ${i + 1}`));

const handleRefresh = async () => {
  console.log("正在刷新...");
  // 模拟网络请求
  await new Promise((resolve) => setTimeout(resolve, 1500));
  // 更新数据
  setItems(["新项目", ...items]);
  console.log("刷新完成");
};

<PullToRefresh
  onRefresh={handleRefresh}
  refreshHeight={70}
  pullThreshold={120}
  refreshingColor="#ff6b6b"
  // 自定义下拉指示器
  pullIndicator={(progress) => (
    <View className="items-center">
      <Text className="text-base font-bold color-red-400">{Math.round(progress * 100)}%</Text>
    </View>
  )}
>
  <View className="px-4">
    {items.map((item, index) => (
      <View key={index} className="p-4 mb-2 rounded-lg bg-gray-50">
        <Text>{item}</Text>
      </View>
    ))}
  </View>
</PullToRefresh>;
```
